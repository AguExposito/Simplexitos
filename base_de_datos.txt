CREATE TABLE Proveedor (
    ID_Proveedor VARCHAR(50) PRIMARY KEY,
    nombreProveedor VARCHAR(100)
);

CREATE TABLE Producto (
    ID_Producto SERIAL PRIMARY KEY,
    codProducto INT,
    nombreProducto VARCHAR(100),
    modeloProducto VARCHAR(100),
    descripcionProducto TEXT
    
);

CREATE TABLE OrdenProveedor (
    ID_OrdenProveedor SERIAL PRIMARY KEY,
    ID_Proveedor VARCHAR(50) REFERENCES Proveedor(ID_Proveedor),
    ID_Producto INT REFERENCES Producto(ID_Producto),
    costoPorUnidad FLOAT,
    costoTotal FLOAT,
    tiempoDeEnvio TIME,
    costoDepositoOrden FLOAT
);

CREATE TABLE Inventario (
    ID_Inventario SERIAL PRIMARY KEY,
    cantidadEnStock INT,
    frecuenciaDeReabastecimiento VARCHAR(50),
    nombreInventario VARCHAR(100),
    costoDepositoInv FLOAT,
    costoVenta FLOAT,
    costoPedido FLOAT,
    CGI VARCHAR(50),
    puntoDeVenta TIME,
	ID_Producto INT REFERENCES Producto(ID_Producto)
);

CREATE TABLE Venta (
    ID_Venta SERIAL PRIMARY KEY,
    cantidadDeVenta INT,
    ID_Inventario INT REFERENCES Inventario(ID_Inventario)
);

CREATE TABLE OrdenDeCompra (
    ID_OrdenDeCompra SERIAL PRIMARY KEY,
    descripcionOrdenDeCompra VARCHAR(255),
    estadoOrdenDeCompra VARCHAR(50),
    ID_Inventario INT REFERENCES Inventario(ID_Inventario)
);
